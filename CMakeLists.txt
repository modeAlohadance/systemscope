cmake_minimum_required(VERSION 3.20)
project(SystemScope VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/W4 /utf-8)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CTest)
add_executable(core_tests tests/core_tests.cpp)
add_test(NAME core COMMAND core_tests)
if(WIN32)
  add_executable(SystemScope WIN32 src/main.cpp)
  target_compile_definitions(SystemScope PRIVATE UNICODE _UNICODE)
  target_link_libraries(SystemScope PRIVATE comctl32 comdlg32)
endif()
